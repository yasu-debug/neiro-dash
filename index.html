async function create() { this.add.image(400, 300, 'background'); this.scoreText = this.add.text(16, 16, '回避: 0 / 10', { fontSize: '32px', fill: '#FFF' }); const rankings = await window.getRanking(); let rankText = "【ランキング TOP3】\n"; rankings.forEach((r, i) => rankText += `${i+1}位: ${r.name} - ${r.score}秒\n`); this.rankingText = this.add.text(16, 60, rankText, { fontSize: '18px', fill: '#000', backgroundColor: '#fff', padding: 5 }); // --- 1. グループを先に作る --- this.obstacles = this.physics.add.group(); this.items = this.physics.add.group(); this.catTraps = []; // --- 2. プレイヤー作成 --- this.player = this.physics.add.sprite(150, 400, 'run').setVisible(false); this.player.body.setAllowGravity(true); this.ground = this.physics.add.staticGroup().create(400, 610, null).setSize(800, 40).setVisible(false); this.physics.add.collider(this.player, this.ground); // --- 3. 衝突判定を追加（重要！） --- this.physics.add.collider(this.player, this.obstacles, () => this.resetGame()); this.physics.add.overlap(this.player, this.items, (player, item) => { item.destroy(); this.bonusTime = (this.bonusTime || 0) + 2000; }); // --- 4. その他のUI作成 --- this.startBtn = this.add.image(400, 350, 'startButton').setInteractive(); const ytIcon = this.add.image(0, 0, 'ytIcon').setScale(0.15); const ytLabel = this.add.text(0, 35, 'YouTubeへ', { fontSize: '20px', fill: '#000', backgroundColor: '#fff', padding: 5 }).setOrigin(0.5); this.ytContainer = this.add.container(400, 500, [ytIcon, ytLabel]).setInteractive(new Phaser.Geom.Rectangle(-50, -50, 100, 100), Phaser.Geom.Rectangle.Contains); this.ytContainer.on('pointerdown', () => window.open('https://www.youtube.com/channel/UC9tdREypdhZQ9Hemq82r95Q', '_blank')); // --- 5. リセットとスタート関数 --- this.resetGame = () => { this.gameStarted = false; this.avoidCount = 0; this.bonusTime = 0; this.scoreText.setText('回避: 0 / 10'); this.player.body.stop(); this.player.setVisible(false).body.enable = false; this.player.setPosition(150, 400); this.startBtn.setVisible(true); this.ytContainer.setVisible(true); this.rankingText.setVisible(true); this.obstacles.clear(true, true); this.items.clear(true, true); this.catTraps = []; this.sound.stopAll(); }; const startGame = () => { if (this.gameStarted) return; this.startBtn.setVisible(false); this.ytContainer.setVisible(false); this.rankingText.setVisible(false); this.player.setVisible(true).body.enable = true; this.player.setPosition(150, 400); this.gameStarted = true; this.startTime = this.time.now; this.sound.play('bgm', { loop: true }); }; this.startBtn.on('pointerdown', startGame); this.input.keyboard.on('keydown-ENTER', startGame); this.time.addEvent({ delay: 1500, callback: spawnObstacle, callbackScope: this, loop: true }); this.cursors = this.input.keyboard.createCursorKeys(); }
